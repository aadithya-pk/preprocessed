\section{Introduction}
\begin{frame}{Introduction}
  \begin{itemize}
    \item Electricity load forecasting predicts future demand from historical consumption and influencing factors such as weather and calendar effects.
    \item Accurate short-term load forecasting (STLF) is vital for power-system planning, smart grids, renewable integration and demand response.
    \item With smart meters, forecasting has moved from statistical methods to machine learning and deep learning.
  \end{itemize}
  \vspace{4pt}
  \begin{block}{Motivation}
    Growing demand, distributed renewables, EVs and smart homes make grids more complex. Forecast errors raise operating costs and reduce grid stability, so accurate and practical short-term forecasts are needed.
  \end{block}
\end{frame}

\section{Problem Statement}
\begin{frame}{Problem Statement}
  \begin{itemize}
    \item Aggregated grid load is \textbf{nonlinear, time-varying and strongly peaked}, which makes accurate forecasting difficult.
    \item Recent deep models (Transformers, mixers, hybrids) are often evaluated on long-horizon benchmarks. It is unclear how they compare with strong tabular baselines on real smart-meter grid data under one common protocol.
    \item Single models suffer from an \textbf{Inductive Bias Dilemma}:
    \begin{itemize}
      \item \textbf{Decision trees (GBDT):} predict piece-wise step functions and lag behind continuous demand ramps;
      \item \textbf{Deep Transformers:} Softmax self-attention acts as a low-pass filter, over-smoothing morning and evening peaks;
      \item \textbf{Operational degradation:} lose accuracy during \textbf{peak hours and cold snaps}, when risk is highest;
      \item \textbf{Compute \& Rigor:} require heavy compute without matching accuracy gains, and rarely come with statistical proof.
    \end{itemize}
  \end{itemize}
  \vspace{3pt}
  \begin{alertblock}{Need}
    An accurate, robust half-hour-ahead forecaster for aggregated London smart-meter load that fuses lagged load, weather and calendar information, and is validated against state-of-the-art models with regime-wise and statistical tests.
  \end{alertblock}
\end{frame}

\section{Research Gap}
\begin{frame}{Research Gaps}
  \begin{block}{Gap 1: Limited head-to-head benchmarking}
    Recent Transformer/mixer models (PatchTST, iTransformer, TimeMixer, EMSformer) are seldom compared with strong tabular baselines such as LightGBM on real aggregated smart-meter data under one protocol.
  \end{block}
  \begin{block}{Gap 2: Heterogeneous hybrids underexplored}
    Tree models capture feature interactions and deep models capture temporal patterns. Few works combine them through a validated convex blend.
  \end{block}
  \begin{block}{Gap 3: Robustness is rarely reported}
    Most studies report overall MAE/RMSE only, not peak-hour, overnight, cold-snap or weekend behaviour.
  \end{block}
\end{frame}

\begin{frame}{Research Gaps (contd.)}
  \begin{block}{Gap 4: Missing statistical validation}
    Improvements are seldom backed by significance tests such as Diebold--Mariano.
  \end{block}
  \begin{block}{Gap 5: Accuracy--efficiency trade-off}
    Advanced hybrids need large compute (hours of training) that limits practical deployment; the trade-off is seldom analysed explicitly.
  \end{block}
\end{frame}

\section{Methodology}
\begin{frame}{Methodology -- Overview}
  \begin{enumerate}
    \item Stream the 5,567-household half-hourly block files and sum energy per timestamp to get \texttt{grid\_load\_kWh}, along with the number of reporting households.
    \item Fill short gaps (time interpolation, at most 4 steps) and drop the rest.
    \item Resample hourly Dark Sky weather to 30 minutes and inner-join it with the grid load.
    \item Engineer 19 input features: weather, calendar, holiday, cyclical, lag and rolling features.
    \item Split chronologically into Train 70\% / Validation 15\% / Test 15\%; fit scalers on train only.
    \item Train the 3 base learners (GBDT, TCN, PatchBiLSTM) and the comparison models; early stopping on validation.
    \item Learn convex ensemble weights on validation, then evaluate once on the held-out test set.
    \item Compare using accuracy metrics, regime stress tests and Diebold--Mariano tests.
  \end{enumerate}
\end{frame}

\begin{frame}{Methodology -- Preprocessing Details}
  \small
  \begin{itemize}
    \item \textbf{Aggregation:} half-hourly \texttt{energy(kWh/hh)} is summed across all households per timestamp. The count of reporting households is kept as \texttt{n\_households}, and per-household load is derived from it.
    \item \textbf{Gap handling:} the series is forced to a continuous 30-minute index; gaps of up to 4 steps (2 h) are time-interpolated and any remaining missing rows are dropped.
    \item \textbf{Weather:} temperature, apparent temperature, humidity and pressure (hourly) are time-interpolated to 30 minutes and inner-joined on timestamp.
    \item \textbf{Warm-up:} the first 7 days are dropped because of the 336-step lag.
    \item \textbf{Outputs:} train/val/test CSVs, a 200-row verification sample and a metadata JSON summary.
  \end{itemize}
\end{frame}

  \small

    \toprule
    \textbf{Group} & \textbf{Features} \\ \midrule

\section{Algorithm}
\begin{frame}{Algorithm}
  \begin{enumerate}
    \item Stream block files; compute aggregated load and household count per 30-minute timestamp.
    \item Interpolate short gaps (at most 4 steps); merge interpolated hourly weather.
    \item Build calendar, holiday, cyclical, lag and rolling features; drop the 7-day warm-up.
    \item Split chronologically into train, validation and test; fit the standard scaler on train only.
    \item For each model, build 48-step input windows and train to predict the next half-hour; use early stopping on validation.
    \item Collect validation predictions of the base learners (GBDT, TCN, PatchBiLSTM).
    \item Solve for convex weights ($w_i\ge0$, $\sum w_i=1$) on validation to form the SuperEnsemble.
    \item Evaluate all models once on the held-out test set; run regime and Diebold--Mariano analyses.
  \end{enumerate}
\end{frame}

\section{Dataset}
\begin{frame}{Dataset -- London Smart Meters (UK Power Networks)}
  \begin{itemize}
    \item 30-minute energy readings (kWh/hh) from \textbf{5,567 households}, stored in block files and aggregated to one grid-level series.
    \item \textbf{Weather (Dark Sky, hourly):} temperature, apparent temperature, humidity, pressure.
    \item \textbf{UK bank holidays} encoded as a binary holiday flag.
    \item Other available tables (daily statistics, half-hour matrix, ACORN demographics) are \emph{not} used.
  \end{itemize}
  \vspace{4pt}
  \begin{alertblock}{Scope}
    Experiments use the London dataset only (Nov 2011 -- Feb 2014). AMPds is not part of the current experiments.
  \end{alertblock}
\end{frame}

\section{Observations}
\begin{frame}{Observations -- Accuracy}
  \small
  \begin{itemize}
    \item The SuperEnsemble is best on every metric: MAE 21.64 kWh, RMSE 28.46 kWh, WAPE 1.80\%, MAPE 1.83\%, R\textsuperscript{2} 0.9950, and a \textbf{13.5\% lower MAE than LightGBM}.
    \item \textbf{LightGBM is a very strong baseline:} with engineered lag, calendar and weather features it beats every standalone deep model, including all published ones and the additional hybrid variants.
    \item No standalone deep or hybrid model surpassed LightGBM. The gain comes from blending complementary learners (trees + deep temporal models).
    \item The best standalone hybrid variant, MSD-TCN (MAE 28.55), improves on the best published baseline, TCN-QRNN (35.38), by about 19\%.
    \item In the ablation, the residual tree stage lowers MS-STLFNet MAE from 44.91 to 43.88.
    \item Transformer-family baselines (iTransformer, EMSformer, PatchTST) perform worst; PatchTST reaches only R\textsuperscript{2} = 0.65. A possible reason is that they are designed for long-horizon forecasting, but this was not tested here.
  \end{itemize}
\end{frame}

\begin{frame}{Observations -- Robustness and Significance}
  \small
  \begin{itemize}
    \item The proposed model has the lowest error in every regime: peak hours 23.39 (LightGBM 27.93), overnight 11.55 (14.53), cold snaps 19.50 (24.13) and weekends 21.69 (24.01).
    \item Peak hours are where weaker models fail most: EMSformer 159.94, PatchTST 335.25, LTE Hybrid 104.00 and iTransformer 101.57 kWh MAE.
    \item In the peak week, EMSformer overshoots the evening peaks by roughly 150--250 kWh, while the SuperEnsemble follows peaks and valleys without visible lag. Its largest visible miss is the Christmas Day midday plateau, which is an atypical holiday profile.
    \item All eight Diebold--Mariano tests are significant ($p<0.0001$). The win rate against LightGBM is 59.8\%, so that gain is consistent but modest, unlike the large gaps against the deep baselines.
  \end{itemize}
\end{frame}

\begin{frame}{Observations -- Accuracy vs Efficiency}
  \small
  \begin{itemize}
    \item The ensemble's 0.28 s covers only the convex weight optimisation. It does not include training the base learners.
    \item LightGBM reaches the second-best accuracy in 1.6 s, the best accuracy-to-cost ratio among single models.
    \item Several deep models need minutes of training (about 190--450 s) and still trail LightGBM. LTE Hybrid takes 5,672 s with MAE 58.02.
    \item Practical takeaway: a tree-based core plus a small deep component gives the best accuracy at a moderate total cost.
  \end{itemize}
\end{frame}

\section{Limitations and Future Work}
\begin{frame}{Limitations and Future Work}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Limitations}
    \begin{itemize}
      \item Single dataset (London) at aggregated level only
      \item Only $H=1$ (30 min ahead) evaluated
      \item Test period covers winter only
      \item Bank holidays are hard-coded for 2011--2014
      \item Point forecasts only, no uncertainty
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{Future work}
    \begin{itemize}
      \item Multi-step horizons (e.g.\ 6 h, 24 h)
      \item Cross-dataset validation (e.g.\ AMPds, other regions)
      \item Probabilistic (quantile) forecasting
      \item Test across all seasons
      \item Real-time deployment on streaming data
    \end{itemize}
  \end{columns}

% SLIDE 28: Conclusion
% ============================================================
\section{Conclusion}
\begin{frame}{Conclusion}
  \begin{itemize}
    \item We built a leakage-free pipeline that aggregates 5,567 smart meters into a grid-level half-hourly forecasting benchmark with weather and calendar features.
    \item The proposed convex STLF-SuperEnsemble (GBDT + TCN + PatchBiLSTM) achieves MAE 21.64 kWh and R\textsuperscript{2} 0.9950, beating LightGBM by 13.5\% and seven published models by 39--89\% (MAE), with significance at $p<0.0001$ (Diebold--Mariano).
    \item It is also the most robust model at peak hours, overnight, during cold snaps and at weekends.
    \item Strong tabular baselines remain hard to beat for half-hour-ahead load forecasting; hybrid blending of complementary learners is the most effective way we found to improve on them.
  \end{itemize}
\end{frame}
